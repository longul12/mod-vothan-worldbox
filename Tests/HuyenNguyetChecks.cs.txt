using System;
using System.IO;
using System.Collections.Generic;
using Newtonsoft.Json;
namespace Sakuga
{
    public static class HuyenNguyetChecks
    {
        public static void Run(string root)
        {
            if((int)School.HuntersWay!=7 || (int)School.HuyenNguyet!=8 || Schools.Count!=8)
                throw new Exception("Existing school IDs changed");
            var ids=new HashSet<string>();int count=0,novice=0,adept=0,expert=0,master=0,grandmaster=0,god=0,moonWaves=0,safeAreas=0;bool returning=false,fan=false,counter=false,launcher=false,clones=false,explosions=false;
            foreach(string file in Directory.GetFiles(Path.Combine(root,"Forge"),"*.json"))
            {
                var recipes=JsonConvert.DeserializeObject<List<Recipe>>(File.ReadAllText(file));
                foreach(var recipe in recipes)
                {
                    if(!ids.Add(recipe.id))throw new Exception("Duplicate recipe: "+recipe.id);
                    if(recipe.school!="huyen_nguyet")continue;
                    count++;
                    var tech=new ForgedTechnique(recipe,root); // Real parser checks enums and every opcode.
                    if(tech.School!=School.HuyenNguyet || tech.Entry<Tier.Novice || tech.Entry>Tier.God)
                        throw new Exception("Wrong moon school/entry");
                    if(tech.Entry==Tier.Novice)novice++;else if(tech.Entry==Tier.Adept)adept++;else if(tech.Entry==Tier.Expert)expert++;else if(tech.Entry==Tier.Master)master++;else if(tech.Entry==Tier.Grandmaster)grandmaster++;else god++;
                    if(tech.Icon!="ui/Icons/sakuga/iconSchoolHuyenNguyet")throw new Exception("Missing school icon fallback");
                    for(int i=0;i<3;i++)
                        if(recipe.mana[i]<=0 || recipe.mana[i]>130 || recipe.cooldown[i]<4 || recipe.cooldown[i]>65)
                            throw new Exception("Invalid preview costs");
                    if(tech.Entry==Tier.Grandmaster)
                    {
                        if(recipe.mana[0]<32 || recipe.cooldown[0]<11)throw new Exception("Grandmaster cost too low");
                        foreach(string line in recipe.steps)
                        {
                            var step=ForgeStep.Parse(line);
                            if(step.Op=="shot" && step.K("lunar",0)>0)
                            {
                                moonWaves++;
                                if(step.K("count",1)!=1 || step.K("pierce",0)>2 || step.K("decay",1)!=.9f || step.K("floor",1)!=.65f)
                                    throw new Exception("Triple crescent exceeds piercing budget");
                            }
                            if(step.Op=="skybeam" || step.Op=="blackhole")
                            {
                                safeAreas++;
                                if(step.K("terrain",1)!=0 || step.K("dur",0)>1 || step.A(0,0)>3.5f)
                                    throw new Exception("Grandmaster area exceeds terrain/time/radius budget");
                            }
                            if(step.Op=="clones" && step.A(0,0)>4)throw new Exception("Too many base clones");
                            if(step.Op=="shot" && step.K("count",1)>3)throw new Exception("Too many base projectiles");
                        }
                    }
                    if(tech.Entry==Tier.God)
                    {
                        if(recipe.cooldown[0]<24 || recipe.mana[0]<55)throw new Exception("God recipe too cheap");
                        foreach(string line in recipe.steps)
                        {
                            var step=ForgeStep.Parse(line);
                            if(step.Op=="shot" && (step.K("crater",-1)!=0 || ForgeInterpreter.ShotCount(step,1.6f)>9))
                                throw new Exception("God projectile count/terrain budget exceeded");
                        }
                    }
                    if(recipe.id=="hn_hoi_hoan_nguyet_nhan")
                    {
                        foreach(string line in recipe.steps)if(line.Contains("flight=Boomerang"))
                        {
                            var step=ForgeStep.Parse(line);
                            if(step.A(1,0)>0.9f || step.K("pierce",0)>3)throw new Exception("Returning blade exceeds two-pass budget");
                            returning=true;
                        }
                    }
                    if(recipe.id=="hn_ban_nguyet_hoanh_tao")fan=recipe.plan=="self" && recipe.need==2;
                    if(recipe.id=="hn_nguyet_kinh_phan_kich")counter=tech.Reactive && recipe.contact=="parried";
                    if(recipe.id=="hn_thang_nguyet_kich")launcher=tech.Role==Role.Launcher && Array.Exists(recipe.steps,s=>s=="leaves Launched");
                    if(recipe.id=="hn_nguyet_anh_phan_than")clones=Array.Exists(recipe.steps,s=>s=="clones 2 Palm 1.1");
                    if(recipe.id=="hn_nguyet_hoa_lien_bao")
                    {
                        explosions=recipe.plan=="area" && recipe.need==2;
                        foreach(string line in recipe.steps)
                        {
                            var step=ForgeStep.Parse(line);
                            if(step.Op=="shot" && (step.K("crater",-1)!=0 || step.K("count",1)>3 || step.K("area",0)>1.2f))
                                throw new Exception("Explosive volley exceeds terrain/projectile budget");
                        }
                    }
                }
            }
            var moon=new NguyetKhuyetTech();
            if(ids.Contains(moon.Id) || moon.Entry!=Tier.Master || moon.School!=School.HuyenNguyet)throw new Exception("Native moon identity/realm mismatch");
            if(count!=33 || novice!=6 || adept!=6 || expert!=6 || master!=5 || grandmaster!=6 || god!=4)throw new Exception("Expected 33 recipes plus 3 native techniques = 36 forms");
            foreach(Technique ultimate in new Technique[]{new ThienNguyetTruyTheTech(),new VinhDaMinhNguyetTech()})
                if(!ids.Add(ultimate.Id) || ultimate.Entry!=Tier.God || ultimate.School!=School.HuyenNguyet || ultimate.GetType().GetConstructor(Type.EmptyTypes)==null)
                    throw new Exception("Native God technique not discoverable or duplicate");
            if(LunarGodRules.FallProgress(.8f)!=0 || LunarGodRules.FallProgress(2)!=1 || LunarGodRules.ImpactDamage(9)!=3.8f)
                throw new Exception("Moon warning/fall/damage bounds changed");
            if(LunarGodRules.MaxActive!=6 || LunarGodRules.MaxTargets!=24 || LunarGodRules.MaxCandidates!=64 || LunarGodRules.DomainPulses!=4)
                throw new Exception("Lunar God runtime budget changed");
            var pixels=LunarGodHazard.BuildMoonCanvas();
            if(pixels.W!=81 || pixels.H!=81 || pixels.Px[0].a!=0 || pixels.Px[40*81+40].a!=255)
                throw new Exception("Moon canvas transparency/size invalid");
            if(ForgeInterpreter.ShotCount(ForgeStep.Parse("shot Orb 0.23 count=9 fixed=1"),1.6f)!=9 ||
               ForgeInterpreter.ShotCount(ForgeStep.Parse("shot Orb 0.5 count=3"),1.6f)!=5)
                throw new Exception("Fixed nine-orb volley or legacy scaling broken");
            if(moonWaves!=3 || safeAreas!=2)throw new Exception("Missing triple crescent or safe lunar areas");
            foreach(string name in new[]{"SkyBeam","BlackHole"})
                if(!(bool)typeof(ChoreoMove).GetMethod(name).GetParameters()[4].DefaultValue)
                    throw new Exception("Existing area techniques must retain terrain default");
            if(MoonSwordApi.GrantedLevel(true,0)!=1 || MoonSwordApi.GrantedLevel(true,3)!=3 || MoonSwordApi.GrantedLevel(false,0)!=0 || MoonSwordApi.GrantedLevel(false,2)!=2)
                throw new Exception("Artifact grant stacks or erases learned ability");
            if(MoonSwordApi.Multiplier(1)!=1.4f || MoonSwordApi.Cooldown(3)!=14f)throw new Exception("Native moon balance changed");
            var shot=new ShotSpec {LunarCrescent=true,Pierce=7,PierceDecay=.9f,PierceFloor=.65f};
            var copy=(ShotSpec)shot.MemberwiseCloneShot();
            if(!copy.LunarCrescent || copy.Pierce!=7 || copy.PierceDecay!=.9f || copy.PierceFloor!=.65f)
                throw new Exception("Native projectile copy loses moon settings");
            var ordinary=new ShotSpec();
            if(ordinary.LunarCrescent || ordinary.PierceDecay!=1 || ordinary.PierceFloor!=1)throw new Exception("Existing projectiles changed defaults");
            if(!File.Exists(Path.Combine(root,"GameResources/ui/Icons/sakuga/iconHnNguyetKhuyet.png")))throw new Exception("Native moon icon missing");
            if(!launcher || !clones || !explosions)throw new Exception("Missing expert launcher, bounded clones or explosive volley");
            if(!returning || !fan || !counter)throw new Exception("Missing returning blade, crowd sweep or reactive guard");
            if(!File.Exists(Path.Combine(root,"GameResources/ui/Icons/sakuga/iconSchoolHuyenNguyet.png")))
                throw new Exception("Moon icon missing");
        }
    }
}
